Integrasi Google Spreadsheets (Rekap Harian & Bulanan Otomatis)

Hubungkan Web App URL Google Apps Script agar data realisasi langsung masuk ke Google Sheet.

// Tempel di Google Spreadsheet -> Ekstensi -> Apps Script -> Deploy Web App (Akses: Anyone):

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var sheetHarian = ss.getSheetByName("Realisasi_Harian") || ss.insertSheet("Realisasi_Harian");
  if (sheetHarian.getLastRow() === 0) {
    sheetHarian.appendRow(["ID","Tanggal","Bulan","PT","Afdeling","Blok","Pekerjaan","Hasil","Satuan","HK","Material_1","Qty_1","Unit_1","Material_2","Qty_2","Unit_2","Mandor","Asisten","Keterangan"]);
  }
  var bulan = data.tanggal ? data.tanggal.substring(0, 7) : "";
  sheetHarian.appendRow([
    data.id, data.tanggal, bulan, data.pt, data.afdeling, data.blok,
    data.pekerjaan, data.hasil, data.satuan, data.hk,
    data.mat1Nama, data.mat1Qty, data.mat1Unit,
    data.mat2Nama, data.mat2Qty, data.mat2Unit,
    data.mandor, data.asisten, data.keterangan
  ]);
  return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON);
}
